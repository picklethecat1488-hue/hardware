def setup_scope_logging(log_file: Optional[Path] = None, verbose: bool = False) -> logging.Logger:
    """Configure structured logging for Scope workstation."""
    log_format = "%(asctime)s [%(levelname)s] %(name)s: %(message)s"
    date_format = "%Y-%m-%d %H:%M:%S"

    logger = logging.getLogger("scope")
    logger.setLevel(logging.DEBUG if verbose else logging.INFO)

    if not logger.handlers:
        target_log = log_file or (_repo_dir / "build" / "scope.log")
        target_log.parent.mkdir(parents=True, exist_ok=True)
        fh = logging.FileHandler(str(target_log), encoding="utf-8")
        fh.setLevel(logging.DEBUG if verbose else logging.INFO)

            ch = logging.StreamHandler(sys.stderr)
            ch.setLevel(logging.DEBUG)
            ch.setFormatter(logging.Formatter(log_format, date_format))

    return logger


def is_port_in_use(port: int, host: str = "127.0.0.1") -> bool:
    """Check if a network port is already actively bound."""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.settimeout(0.5)
        return s.connect_ex((host, port)) == 0


def find_available_port(start_port: int = 8000, host: str = "127.0.0.1", max_attempts: int = 50) -> int:
    """Find the first open TCP port beginning at start_port."""
    for p in range(start_port, start_port + max_attempts):
        if not is_port_in_use(p, host):
            return p
    return start_port

def parse_target_string(target_str: Optional[str]) -> tuple[str, str, Optional[str], str]:
    """Parse project, target, subassembly, and visualization mode from target string."""
    if not target_str:
        return "", "", None, "CAD"

    mode = "CAD"
    clean_target = target_str

    if clean_target.endswith(":pcb") or "/pcb" in clean_target:
        mode = "PCB"
        clean_target = clean_target.replace(":pcb", "").replace("/pcb", "")
    elif clean_target.endswith(":simulate") or "/simulate" in clean_target:
        mode = "SIMULATE"
        clean_target = clean_target.replace(":simulate", "").replace("/simulate", "")
    elif clean_target.endswith(":diagram") or "/diagram" in clean_target:
        mode = "DIAGRAM"
        clean_target = clean_target.replace(":diagram", "").replace("/diagram", "")

    # Strip action specifiers like :view or :part
    if clean_target.endswith(":view") or "/view" in clean_target:
        clean_target = clean_target.replace(":view", "").replace("/view", "")
    elif clean_target.endswith(":part") or "/part" in clean_target:
        clean_target = clean_target.replace(":part", "").replace("/part", "")

    subassembly = None
    if ":" in clean_target:
        clean_target, subassembly = clean_target.split(":", 1)

    base = clean_target.split(":", 1)[0]
    project = base.split("/", 1)[0] if "/" in base else "carrier_board"
    target = base.split("/", 1)[1] if "/" in base else base

    # Intelligent target mapping for simulation targets without separate view definitions
    if project == "cat_fountain" and target == "simulation":
        target = "product"
        mode = "SIMULATE"

    return project, target, subassembly, mode

                except (json.JSONDecodeError, OSError):
                    pass
            time.sleep(0.05)

        print(f"➜ Scope workstation running at {server_url} (PID {proc.pid}).")
        print("➜ Detached from terminal.")
        sys.exit(0)

    # Foreground / non-detached server process
    scope_logger = setup_scope_logging(args.log_file, verbose=args.verbose)
    scope_logger.info("Starting Scope workstation in foreground (PID %d)...", os.getpid())

    lock_file.parent.mkdir(parents=True, exist_ok=True)
    lock_file.write_text(
        json.dumps(
            {
                "pid": os.getpid(),
                "url": server_url,
                "no_browser": getattr(args, "no_browser", False),
            }
        ),
        encoding="utf-8",
    )

    print(f"➜ Scope workstation running at {server_url} (PID {os.getpid()}).")

    from provider.scope import ScopeServer, ScopeGeometryPipeline

    pipeline = ScopeGeometryPipeline()
    server = ScopeServer(
        host=args.host,
        port=port,
        repo_root=_repo_dir,
        pipeline=pipeline,
        logger=scope_logger,
        bind_and_activate=True,
    )
    if proj and target:
        server.warm_up(
            project=proj,
            target=target,
            subassembly=sub,
            mode=mode,
        )

    if not args.no_browser:

        def _on_close():
            print("Desktop window closed; shutting down scope server...")
            threading.Thread(target=server.shutdown, daemon=True).start()

        threading.Thread(
            target=launch_browser,
            args=(server_url, args.browser, _on_close),
            daemon=True,
        ).start()

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down scope server...")
    finally:
        if lock_file.exists():
            try:
                lock_file.unlink()
            except OSError:
                pass
        server.server_close()


if __name__ == "__main__":
    main()
